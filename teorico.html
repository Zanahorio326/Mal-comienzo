import { getAuth, deleteUser } from "https://www.gstatic.com/firebasejs/11.4.0/firebase-auth.js"; import { getDatabase, ref, remove } from "https://www.gstatic.com/firebasejs/11.4.0/firebase-database.js"; // Función para eliminar cuenta y datos function eliminarCuenta() { const auth = getAuth(); const db = getDatabase(); const user = auth.currentUser; if (user) { // Eliminar datos del usuario en la base de datos remove(ref(db, 'usuarios/' + user.uid)) .then(() => { console.log('Datos eliminados de la base de datos'); // Eliminar cuenta de Firebase Authentication deleteUser(user) .then(() => { console.log('Cuenta eliminada con éxito'); // Redirigir o actualizar UI window.location.href = '/'; // Redirigir a la página de inicio o login }) .catch((error) => { console.error('Error al eliminar la cuenta:', error); }); }) .catch((error) => { console.error('Error al eliminar los datos de la base de datos:', error); }); } else { console.log('No hay usuario autenticado'); } }
